การเชื่อมต่อ Google Sheets API

คุณสามารถเชื่อมต่อระบบเข้ากับ Google Sheets ผ่าน Google Apps Script เพื่อให้ทุกการส่งงานบันทึกลงสเปรดชีตของคุณได้แบบ Real-time

คำแนะนำและโค้ด Google Apps Script (GS Code)

ก๊อปปี้โค้ดด้านล่างนี้ไปวางใน Google Apps Script ใน Google Sheet ของคุณ:

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  sheet.appendRow([
    new Date(),
    data.studentId,
    data.studentName,
    data.room,
    data.activityTitle,
    data.fileUrl,
    data.status || 'รอตรวจ',
    data.score || '-',
    data.feedback || ''
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
